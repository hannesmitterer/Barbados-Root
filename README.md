### Browser Usage

Include the modules in your HTML:

```html
<script src="euystacio-helmi-ai.js"></script>
<script src="consciousness-kernel.js"></script>
<script src="ggi-aic.js"></script>
<script src="barbados-integration.js"></script>
```

Initialize the integrated system:

```javascript
const barbados = new BarbadosRootIntegration({
    autoActivate: true,
    barbadosNodeId: 'BARBADOS_ROOT_SOVEREIGNTY'
});

// Get system status
const status = barbados.getSystemStatus();
console.log('Operational Status:', status.operationalStatus);
```
